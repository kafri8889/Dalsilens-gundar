class MainActivity : ComponentActivity() {

    private val requestPermissionLauncher = registerForActivityResult(
        ActivityResultContracts.RequestPermission()
    ) { isGranted: Boolean ->
        if (isGranted) { hasCameraPermission = true }
        else Toast.makeText(this,
            "Camera permission is required", Toast.LENGTH_SHORT).show()

    private var hasCameraPermission by mutableStateOf(false)

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        hasCameraPermission = ContextCompat.checkSelfPermission(
            this, Manifest.permission.CAMERA
        ) == PackageManager.PERMISSION_GRANTED

        setContent {
            DaltonLensTheme {
                val navController = rememberNavController()
                NavHost(navController = navController,
                        startDestination = "dashboard") {
                    composable("dashboard") {
                        DashboardScreen(
                            onNavigateToCamera = {
                                if (hasCameraPermission)
                                    navController.navigate("camera")
                                else
                                    requestPermissionLauncher
                                        .launch(Manifest.permission.CAMERA)
                            },

                                if (hasCameraPermission)
                                    navController.navigate("color_classification")
                                else
                                    requestPermissionLauncher
                                        .launch(Manifest.permission.CAMERA)
                            }
                        )
                    }
                    composable("camera") { CameraScreen() }
                    composable("color_classification") {
                        ColorClassificationTestScreen()
                    }
                }
            }
        }
    }
}

\subsection*{B. ColorProcessor.kt}

\begin{lstlisting}[caption={Implementasi konversi ruang warna RGB--LMS dan pembangkitan matriks transformasi defisiensi serta daltonisasi}, label={lst:colorprocessor}]
enum class DeficiencyType(val displayName: String) {
    Normal("Normal"), Protanopia("Protanopia"),
    Deuteranopia("Deuteranopia"), Tritanopia("Tritanopia"), Custom("Custom")
}

object ColorProcessor {

    // Konversi RGB ke LMS menggunakan matriks Hunt-Pointer-Estevez
    fun rgbToLms(r: Float, g: Float, b: Float): Triple<Float, Float, Float> {
        val l = 17.8824f * r + 43.5161f * g + 4.11935f * b
        val m =  3.4565f * r + 27.1554f * g + 3.86714f * b
        val s = 0.02996f * r + 0.18431f * g + 1.46709f * b
        return Triple(l, m, s)
    }

    // Simulasi defisiensi warna dalam ruang LMS
    fun simulateLms(l: Float, m: Float, s: Float,
                    deficiency: DeficiencyType): Triple<Float, Float, Float> {
        return when (deficiency) {
            DeficiencyType.Protanopia   ->
                Triple(2.02344f * m - 2.52581f * s, m, s)
            DeficiencyType.Deuteranopia ->
                Triple(l, 0.494207f * l + 1.24827f * s, s)
            DeficiencyType.Tritanopia   ->
                Triple(l, m, -0.395913f * l + 0.801109f * m)
            else -> Triple(l, m, s)
        }
    }

    // Konversi balik LMS ke RGB (invers matriks Hunt-Pointer-Estevez)
    fun lmsToRgb(l: Float, m: Float, s: Float): Triple<Float, Float, Float> {
        val r =  0.080944f    * l - 0.130504f   * m + 0.116721f  * s
        val g = -0.0102485f   * l + 0.0540194f  * m - 0.113615f  * s
        val b = -0.000365294f * l - 0.00412163f * m + 0.693513f  * s
        return Triple(r, g, b)
    }

    // Bangun matriks 4x5 untuk simulasi defisiensi warna
    fun getDeficiencyMatrix(deficiency: DeficiencyType,
                            severity: Float = 1.0f): FloatArray? {
        if (deficiency == DeficiencyType.Custom) return null
        if (deficiency == DeficiencyType.Normal || severity <= 0.0f) {
            return floatArrayOf(1f,0f,0f,0f,0f, 0f,1f,0f,0f,0f,
                                0f,0f,1f,0f,0f, 0f,0f,0f,1f,0f)
        }
        fun transform(r: Float, g: Float, b: Float): Triple<Float,Float,Float> {
            val (lN,mN,sN) = rgbToLms(r, g, b)
            val (lD,mD,sD) = simulateLms(lN, mN, sN, deficiency)
            return lmsToRgb(lN+(lD-lN)*severity, mN+(mD-mN)*severity,
                            sN+(sD-sN)*severity)
        }
        val (rR,rG,rB) = transform(1f,0f,0f)
        val (gR,gG,gB) = transform(0f,1f,0f)
        val (bR,bG,bB) = transform(0f,0f,1f)
        return floatArrayOf(rR,gR,bR,0f,0f, rG,gG,bG,0f,0f,
                            rB,gB,bB,0f,0f, 0f,0f,0f,1f,0f)
    }

    // Bangun matriks 4x5 untuk koreksi daltonisasi
    fun getDaltonizationMatrix(deficiency: DeficiencyType,
                               severity: Float = 1.0f): FloatArray? {
        if (deficiency == DeficiencyType.Custom ||
            deficiency == DeficiencyType.Normal) {
            return floatArrayOf(1f,0f,0f,0f,0f, 0f,1f,0f,0f,0f,
                                0f,0f,1f,0f,0f, 0f,0f,0f,1f,0f)
        }
        fun transform(rA: Float, gA: Float, bA: Float): Triple<Float,Float,Float> {
            val (l,m,s)   = rgbToLms(rA, gA, bA)
            val (lS,mS,sS)= simulateLms(l, m, s, deficiency)
            val (rS,gS,bS)= lmsToRgb(lS, mS, sS)
            val (eR,eG,eB)= Triple(rA-rS, gA-gS, bA-bS)
            val (sR,sG,sB) = when (deficiency) {
                DeficiencyType.Protanopia   ->
                    Triple(0f, .9f*eR+.1f*eG, .9f*eR+.1f*eB)
                DeficiencyType.Deuteranopia ->
                    Triple(eR+.7f*eG, 0f, .7f*eG+eB)
                DeficiencyType.Tritanopia   ->
                    Triple(eR+.7f*eB, eG+.7f*eB, 0f)
                else -> Triple(0f, 0f, 0f)
            }
            return Triple(rA+sR*severity, gA+sG*severity, bA+sB*severity)
        }
        val (rR,rG,rB) = transform(1f,0f,0f)
        val (gR,gG,gB) = transform(0f,1f,0f)
        val (bR,bG,bB) = transform(0f,0f,1f)
        return floatArrayOf(rR,gR,bR,0f,0f, rG,gG,bG,0f,0f,
                            rB,gB,bB,0f,0f, 0f,0f,0f,1f,0f)
    }
}

\subsection*{C. ColorNames.kt}

\begin{lstlisting}[caption={Implementasi klasifikasi nama warna berbasis ruang warna HSV dengan pendekatan \textit{rule-based thresholding}}, label={lst:colornames}]
class ColorNames(private val context: Context) {

    // Rata-rata piksel dalam area radius x radius di sekitar titik pusat
    fun getAverageColor(bitmap: Bitmap, centerX: Int, centerY: Int,
                        radius: Int = 5): Triple<Int, Int, Int> {
        var r = 0L; var g = 0L; var b = 0L; var count = 0
        for (dx in -radius..radius) {
            for (dy in -radius..radius) {
                val x = (centerX + dx).coerceIn(0, bitmap.width - 1)
                val y = (centerY + dy).coerceIn(0, bitmap.height - 1)
                val pixel = bitmap.getPixel(x, y)
                r += (pixel shr 16) and 0xFF
                g += (pixel shr 8)  and 0xFF
                b +=  pixel         and 0xFF
                count++
            }
        }
        return Triple((r/count).toInt(), (g/count).toInt(), (b/count).toInt())
    }

    // Klasifikasi nama warna dari nilai RGB menggunakan ruang warna HSV
    fun getColorNameFromRGB(r: Int, g: Int, b: Int): String {
        val hsv = FloatArray(3)
        android.graphics.Color.RGBToHSV(r, g, b, hsv)
        val h = hsv[0]  // Hue        : 0.0 - 360.0
        val s = hsv[1]  // Saturation : 0.0 - 1.0
        val v = hsv[2]  // Value      : 0.0 - 1.0

        // Deteksi warna akromatik (hitam, abu-abu, putih) berdasarkan saturasi
        if (s < 0.10f) {
            return when {
                v < 0.15f -> "Black"
                v < 0.40f -> "Dark Gray"
                v < 0.78f -> "Gray"
                else      -> "White"
            }
        }
        if (v < 0.08f) return "Black"          // terlalu gelap
        if (s < 0.18f && v > 0.72f) return "White"  // hampir putih

        // Klasifikasi warna kromatik berdasarkan nilai Hue
        return when {
            h < 9.85f   -> "Red"
            h < 26.98f  -> "Orange"
            h < 50.42f  -> "Yellow"
            h < 122.50f -> "Green"
            h < 204.85f -> "Cyan"
            h < 224.57f -> "Blue"
            h < 319.37f -> "Purple"
            else        -> "Red"
        }
    }

    fun getCenterColorName(bitmap: Bitmap, radius: Int = 5): String {
        val (r, g, b) = getAverageColor(bitmap,
            bitmap.width / 2, bitmap.height / 2, radius)
        return getColorNameFromRGB(r, g, b)
    }
}

data class ColorItem(val hex: String, val name: String)

\begin{lstlisting}[caption={Inisialisasi CameraX dan loop analisis frame secara \textit{real-time}}, label={lst:camerascreen}]
// Inisialisasi CameraX dan pipeline ImageAnalysis
LaunchedEffect(Unit) {
    val cameraProvider = context.getCameraProvider()
    val resolutionSelector = ResolutionSelector.Builder()
        .setAspectRatioStrategy(
            AspectRatioStrategy.RATIO_16_9_FALLBACK_AUTO_STRATEGY)
        .build()

    val preview = Preview.Builder()
        .setResolutionSelector(resolutionSelector).build()
        .also { it.setSurfaceProvider(previewView.surfaceProvider) }

    val imageAnalysis = ImageAnalysis.Builder()
        .setResolutionSelector(resolutionSelector)
        .setBackpressureStrategy(
            ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST)
        .build().also {
            it.setAnalyzer(analysisExecutor) { imageProxy ->
                try {
                    val bitmap = imageProxy.toBitmap()
                    val rotation = imageProxy.imageInfo.rotationDegrees
                    val finalBitmap = if (rotation != 0) {
                        val matrix = Matrix().apply {
                            postRotate(rotation.toFloat())
                        }
                        Bitmap.createBitmap(bitmap, 0, 0,
                            bitmap.width, bitmap.height, matrix, true)
                    } else bitmap

                    // Deteksi nama warna dari piksel tengah frame
                    centerColorName = colorNames
                        .getCenterColorName(finalBitmap, sampleAreaSize)
                    processedBitmap = finalBitmap
                } catch (e: Exception) {
                    Log.e("CameraScreen", "Error processing frame", e)
                } finally {
                    imageProxy.close()
                }
            }
        }

    cameraProvider.unbindAll()
    cameraProvider.bindToLifecycle(
        lifecycleOwner,
        CameraSelector.DEFAULT_BACK_CAMERA,
        preview,
        imageAnalysis
    )
}
